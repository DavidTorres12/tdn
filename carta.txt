// ---- Scroll Reveal ----
(function initReveal() {
    const els = document.querySelectorAll('.reveal-carta');
    if (!('IntersectionObserver' in window)) {
        els.forEach(el => el.classList.add('visible'));
        return;
    }
    const obs = new IntersectionObserver(entries => {
        entries.forEach((e, i) => {
            if (e.isIntersecting) {
                setTimeout(() => e.target.classList.add('visible'), i * 80);
                obs.unobserve(e.target);
            }
        });
    }, { threshold: 0.08, rootMargin: '0px 0px -30px 0px' });
    els.forEach(el => obs.observe(el));
})();

// ---- Sticky Nav - Category Scroll ----
document.querySelectorAll('.carta-nav-btn').forEach(btn => {
    btn.addEventListener('click', () => {
        // Update active state
        document.querySelectorAll('.carta-nav-btn').forEach(b => b.classList.remove('active'));
        btn.classList.add('active');

        // Scroll to section
        const target = document.getElementById(btn.dataset.target);
        if (target) {
            const navH = document.getElementById('cartaNav').offsetHeight;
            const y = target.getBoundingClientRect().top + window.pageYOffset - navH - 10;
            window.scrollTo({ top: y, behavior: 'smooth' });
        }
    });
});

// ---- Auto-update nav on scroll ----
const sections = document.querySelectorAll('.carta-content > section');
const navBtns = document.querySelectorAll('.carta-nav-btn');
window.addEventListener('scroll', () => {
    const navH = document.getElementById('cartaNav') ? document.getElementById('cartaNav').offsetHeight + 20 : 80;
    let currentId = '';
    sections.forEach(sec => {
        if (sec.getBoundingClientRect().top <= navH + 60) {
            currentId = sec.id;
        }
    });
    navBtns.forEach(btn => {
        btn.classList.toggle('active', btn.dataset.target === currentId);
    });
}, { passive: true });

// ---- Back to top ----
const btnTop = document.getElementById('btn-top');
if (btnTop) {
    window.addEventListener('scroll', () => {
        if (window.scrollY > 300) {
            btnTop.classList.add('visible');
        } else {
            btnTop.classList.remove('visible');
        }
    }, { passive: true });

    btnTop.addEventListener('click', () => {
        window.scrollTo({ top: 0, behavior: 'smooth' });
    });
}
